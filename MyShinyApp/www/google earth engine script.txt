// Define constant values for wavelengths
var l3 = 560; // Green wavelength (nm)
var l4 = 665; // Red wavelength (nm)
var l8 = 842; // Near-infrared wavelength (nm)
// Define the function to calculate CI
var calculateCI = function(image) {
  var CI = image.expression(
'- B4 + B3 - (B3 - B8) * (l4 - l3) / (l8 - l3)', {
  'B3': image.select('B3'), // Green band
  'B4': image.select('B4'), // Red band
  'B8': image.select('B8'), // Near-infrared band
  'l3': l3,
'l4': l4,
'l8': l8
});
return CI.rename('CI').clip(NI);
};

// Define constant values for wavelengths
var L4 = 665; // Red wavelength (nm)
var L6 = 740; // Red-edge 2 (nm)
var L8A = 865; // Near Infrared-narrow (nm)
var calculateABDI = function(image) {
  var ABDI = image.expression(
'(B6 - B4 -(B8A - B4) * (L6-L4) / (L8A-L4)) - (B4 - 0.5 * B3)', {
  'B3': image.select('B3'), // Green band
  'B4': image.select('B4'), // Red band
  'B6': image.select('B6'), // Red-edge 2
  'B8A': image.select('B8A'), // Near-infrared-narrow
  'L4': L4,
  'L6': L6,
  'L8A': L8A
});
return ABDI.rename('ABDI').clip(NI);
};
